# =============================================================================
# requirements.txt — Speed Trap / AGD 307 Radar  (Raspberry Pi 5 aarch64)
# Python 3.11 — Raspberry Pi OS Bookworm (Debian 12) 64-bit
# =============================================================================
#
# INSTALL ORDER MATTERS — always run install.sh, which installs torch first.
# If you pip-install manually, follow this order:
#   1. pip install torch==2.3.0 torchvision==0.18.0
#   2. pip install -r requirements.txt
#
# NOTE: picamera2 must be installed as a SYSTEM apt package (NOT via pip):
#   sudo apt-get install python3-picamera2
# Create the venv with --system-site-packages so it stays accessible.
# =============================================================================

# ── Serial communication ──────────────────────────────────────────────────────
pyserial==3.5

# ── Numerical core ────────────────────────────────────────────────────────────
# numpy is pinned to 1.24.4 for TWO independent reasons:
#
#   (a) ABI match: python3-picamera2 is installed via apt and its libcamera
#       Python bindings are compiled against Debian Bookworm's system numpy
#       (1.24.x).  The venv is created with --system-site-packages, so numpy
#       1.26 in the venv can shadow the system numpy.  When picamera2 passes
#       a libcamera buffer as a numpy array into the venv, the C-API struct
#       sizes differ (numpy uses its own dtype ABI), causing either a
#       RuntimeWarning: "numpy.dtype size changed" or a hard segfault on
#       some Bookworm builds.  Using 1.24.4 keeps the ABI identical to the
#       system package.
#
#   (b) ultralytics upper-bound: ultralytics 8.2.x declares numpy<2.0.0 —
#       satisfied by 1.24.4.
#
# Do NOT upgrade to 1.25, 1.26, or 2.x in this environment.
numpy==1.24.4

# ── Image I/O ─────────────────────────────────────────────────────────────────
Pillow==10.3.0              # ultralytics>=7.1.2, scikit-image>=9.1.0 — both satisfied

# ── Computer vision ───────────────────────────────────────────────────────────
# 4.10.0.84 is the version pulled automatically by easyocr 1.7.1 on Linux;
# it also satisfies ultralytics >=4.6.0.  The -headless variant avoids
# pulling libGL.so / X11 deps that are unavailable in a headless Pi setup.
opencv-python-headless==4.10.0.84

# ── Deep learning (install these BEFORE easyocr / ultralytics) ───────────────
# torch 2.3.x ships an official manylinux aarch64 cp311 wheel — no custom
# build needed on Raspberry Pi OS Bookworm.
# Compatibility matrix (from ultralytics/utils/checks.py):
#   torch 2.3  ↔  torchvision 0.18
torch==2.3.0
torchvision==0.18.0

# ── YOLOv8 object detection ───────────────────────────────────────────────────
ultralytics==8.2.18        # last release in 8.2.x series; numpy<2 constraint respected

# ── OCR engine + ALL required direct dependencies ────────────────────────────
# easyocr's declared Requires (pip show easyocr):
#   ninja, numpy, opencv-python-headless, Pillow, pyclipper,
#   python-bidi, PyYAML, scikit-image, scipy, Shapely, torch, torchvision
# Pinning them prevents pip from silently pulling incompatible latest versions.
easyocr==1.7.1
python-bidi==0.4.2         # right-to-left text support (required by easyocr)
pyclipper==1.3.0.post5     # polygon clipping (required by easyocr)
ninja==1.11.1.1             # JIT build backend (required by easyocr)

# ── Scientific stack ──────────────────────────────────────────────────────────
scipy==1.13.0               # requires numpy>=1.22.4 — satisfied by 1.24.4 above
# scikit-image 0.22.0 avoids the imageio>=2.33 pull-in that 0.23.x introduces;
# 0.22.x was compiled against numpy 1.x ABI and works cleanly here.
scikit-image==0.22.0

# ── Shared transitive dependencies ────────────────────────────────────────────
# Pinned to prevent silent upgrades that could break the numpy/torch ABI.
Shapely==2.0.5              # geometry ops; used by both easyocr and ultralytics
PyYAML==6.0.1               # config parsing; used by ultralytics and easyocr
tqdm==4.66.4
requests==2.32.3
matplotlib==3.8.4           # ultralytics plotting (>=3.3.0); 3.9.x also works
seaborn==0.13.2             # ultralytics plotting (>=0.11.0)
pandas==2.2.2               # ultralytics data handling (>=1.1.4)
